// Release Governor — Agent 1
// Version: 3.0.0
// Responsibility: Decide whether a release is allowed to advance

import { EventEmitter } from 'events';
import { createHash } from 'crypto';

// ============================================================
// TYPES — RELEASE STATE
// ============================================================

export type GateStatus = 
  | 'PASS'
  | 'HOLD'
  | 'BLOCK'
  | 'UNVERIFIED'
  | 'ROLLBACK_REQUIRED';

export type ProvenanceClass = 
  | 'SIMULATED'
  | 'RECONSTRUCTED'
  | 'MEASURED'
  | 'DERIVED'
  | 'REFERENCE'
  | 'UNKNOWN';

export type GateStage =
  | 'DRAFT'
  | 'INTAKE'
  | 'DEVELOPMENT'
  | 'INTEGRATION'
  | 'SECURITY'
  | 'STAGING'
  | 'SOFTWARE_READY'
  | 'HARDWARE_PRECHECK'
  | 'H1'
  | 'H2'
  | 'H3'
  | 'H4'
  | 'H5'
  | 'H6'
  | 'HARDWARE_READY'
  | 'LIVE_PRECHECK'
  | 'LIVE'
  | 'PARITY'
  | 'RELEASE_CANDIDATE'
  | 'FINAL_REVIEW'
  | 'AUTHORIZED'
  | 'DEPLOYED'
  | 'MONITORED';

export interface Gate {
  id: string;
  stage: GateStage;
  status: GateStatus;
  owner: string;
  timestamp: string;
  commitSha: string;
  buildId: string;
  testId?: string;
  deviceId?: string;
  environment: string;
  provenance: ProvenanceClass;
  measurements?: Record<string, any>;
  expectedValues?: Record<string, any>;
  actualValues?: Record<string, any>;
  tolerance?: number;
  evidenceUri?: string;
  reviewer?: string;
  decision?: string;
}

export interface ReleaseState {
  version: string;
  gates: Map<string, Gate>;
  overallStatus: GateStatus;
  provenanceRequired: boolean;
  hardwareControl: 'ENABLED' | 'DISABLED';
  rawMediaPersistence: 'ENABLED' | 'DISABLED';
  manifest: ReleaseManifest;
}

export interface ReleaseManifest {
  version: string;
  software: Record<string, GateStatus>;
  hardware: Record<string, GateStatus>;
  live: Record<string, GateStatus>;
  controls: {
    hardware_control: 'ENABLED' | 'DISABLED';
    raw_media_persistence: 'ENABLED' | 'DISABLED';
  };
  overall: {
    status: GateStatus;
    provenance_required: boolean;
  };
  signature?: string;
}

// ============================================================
// EVIDENCE INTERFACE
// ============================================================

export interface Evidence {
  id: string;
  release: string;
  gate: string;
  status: GateStatus;
  provenance: ProvenanceClass;
  deviceId?: string;
  testId?: string;
  measurementRef?: string;
  evidenceHash: string;
  timestamp: string;
  signedBy: string;
}

export interface Incident {
  id: string;
  release: string;
  gate: string;
  severity: 'CRITICAL' | 'HIGH' | 'MEDIUM' | 'LOW';
  observedFailure: string;
  expectedBehavior: string;
  evidence: any;
  environment: string;
  commit: string;
  device?: string;
  probableCause?: string;
  remediation?: string;
  owner: string;
  retestRequired: boolean;
  rollbackRequired: boolean;

// ============================================================
// RELEASE GOVERNOR — AGENT 1
// ============================================================

export class ReleaseGovernor extends EventEmitter {
  private state: ReleaseState;
  private evidenceStore: Map<string, Evidence> = new Map();
  private incidentStore: Map<string, Incident> = new Map();

  constructor(version: string = '3.0.0') {
    super();
    
    this.state = {
      version,
      gates: new Map(),
      overallStatus: 'UNVERIFIED',
      provenanceRequired: true,
      hardwareControl: 'DISABLED',
      rawMediaPersistence: 'DISABLED',
      manifest: this.createInitialManifest(version)
    };
  }

  private createInitialManifest(version: string): ReleaseManifest {
    return {
      version,
      software: {
        development: 'UNVERIFIED',
        integration: 'UNVERIFIED',
        security: 'UNVERIFIED',
        staging: 'UNVERIFIED'
      },
      hardware: {
        H1_sensor: 'UNVERIFIED',
        H2_thermal: 'UNVERIFIED',
        H3_optical: 'UNVERIFIED',
        H4_acquisition: 'UNVERIFIED',
        H5_fpga_hil: 'UNVERIFIED',
        H6_reconstruction_parity: 'UNVERIFIED'
      },
      live: {
        device_handshake: 'UNVERIFIED',
        telemetry: 'UNVERIFIED',
        orchestrator: 'UNVERIFIED',
        dashboard: 'UNVERIFIED',
        digital_twin: 'UNVERIFIED',
        audit_event: 'UNVERIFIED'
      },
      controls: {
        hardware_control: 'DISABLED',
        raw_media_persistence: 'DISABLED'
      },
      overall: {
        status: 'UNVERIFIED',
        provenance_required: true
      }
    };
  }

  // ============================================================
  // GATE MANAGEMENT
  // ============================================================

  public registerGate(gate: Gate): void {
    this.state.gates.set(gate.id, gate);
    this.emit('gateRegistered', gate);
  }

  public getGate(id: string): Gate | undefined {
    return this.state.gates.get(id);
  }

  public updateGate(id: string, status: GateStatus, evidence?: Evidence): void {
    const gate = this.state.gates.get(id);
    if (!gate) {
      throw new Error(`Gate ${id} not found`);
    }

    gate.status = status;
    gate.timestamp = new Date().toISOString();

    if (evidence) {
      this.evidenceStore.set(evidence.id, evidence);
      gate.evidenceUri = evidence.id;
    }

    this.state.gates.set(id, gate);
    this.emit('gateUpdated', { gate, evidence });

    // Check overall status
    this.updateOverallStatus();
  }

  private updateOverallStatus(): void {
    const gates = Array.from(this.state.gates.values());
    
    if (gates.some(g => g.status === 'BLOCK')) {
      this.state.overallStatus = 'BLOCK';
    } else if (gates.some(g => g.status === 'HOLD')) {
      this.state.overallStatus = 'HOLD';
    } else if (gates.some(g => g.status === 'UNVERIFIED')) {
      this.state.overallStatus = 'UNVERIFIED';
    } else if (gates.every(g => g.status === 'PASS')) {
      this.state.overallStatus = 'PASS';
    } else {
      this.state.overallStatus = 'HOLD';
    }

    this.emit('overallStatusUpdated', this.state.overallStatus);
  }

  // ============================================================
  // EVIDENCE VALIDATION
  // ============================================================

  public validateEvidence(evidence: Evidence): boolean {
    // 1. Check provenance
    if (evidence.provenance === 'UNKNOWN') {
      this.emit('evidenceRejected', { evidence, reason: 'UNKNOWN provenance' });
      return false;
    }

    // 2. Check hash
    const computedHash = this.computeEvidenceHash(evidence);
    if (computedHash !== evidence.evidenceHash) {
      this.emit('evidenceRejected', { evidence, reason: 'Hash mismatch' });
      return false;
    }

    // 3. Check gate-specific requirements
    const gate = this.state.gates.get(evidence.gate);
    if (!gate) {
      this.emit('evidenceRejected', { evidence, reason: 'Gate not found' });
      return false;
    }

    // 4. Check physical evidence rules
    if (this.isHardwareGate(gate.stage) && evidence.provenance !== 'MEASURED') {
      this.emit('evidenceRejected', { 
        evidence, 
        reason: 'Hardware gate requires MEASURED provenance' 
      });
      return false;
    }

    // 5. Check live gate rules
    if (this.isLiveGate(gate.stage) && evidence.provenance !== 'MEASURED') {
      this.emit('evidenceRejected', { 
        evidence, 
        reason: 'Live gate requires MEASURED provenance' 
      });
      return false;
    }

    this.emit('evidenceAccepted', evidence);
    return true;
  }

  private computeEvidenceHash(evidence: Evidence): string {
    const data = JSON.stringify({
      id: evidence.id,
      release: evidence.release,
      gate: evidence.gate,
      provenance: evidence.provenance,
      deviceId: evidence.deviceId,
      testId: evidence.testId,
      timestamp: evidence.timestamp
    });
    return createHash('sha256').update(data).digest('hex');
  }

  private isHardwareGate(stage: GateStage): boolean {
    return ['H1', 'H2', 'H3', 'H4', 'H5', 'H6'].includes(stage);
  }

  private isLiveGate(stage: GateStage): boolean {
    return ['LIVE', 'PARITY'].includes(stage);
  }

  // ============================================================
  // INCIDENT MANAGEMENT
  // ============================================================

  public createIncident(incident: Incident): void {
    this.incidentStore.set(incident.id, incident);
    this.emit('incidentCreated', incident);
  }

  public resolveIncident(id: string, resolution: string): void {
    const incident = this.incidentStore.get(id);
    if (!incident) {
      throw new Error(`Incident ${id} not found`);
    }

    incident.remediation = resolution;
    this.incidentStore.set(id, incident);
    this.emit('incidentResolved', incident);
  }

  // ============================================================
  // RELEASE AUTHORIZATION
  // ============================================================

  public authorizeRelease(): boolean {
    // Check all gates
    const gates = Array.from(this.state.gates.values());
    const blocked = gates.filter(g => g.status === 'BLOCK');
    const unverified = gates.filter(g => g.status === 'UNVERIFIED');

    if (blocked.length > 0) {
      this.emit('releaseBlocked', { reason: `${blocked.length} gates blocked` });
      return false;
    }

    if (unverified.length > 0) {
      this.emit('releaseHeld', { reason: `${unverified.length} gates unverified` });
      return false;
    }

    const allPass = gates.every(g => g.status === 'PASS');
    if (!allPass) {
      this.emit('releaseHeld', { reason: 'Not all gates passed' });
      return false;
    }

    // Generate manifest
    this.state.manifest = this.generateManifest();
    this.state.overallStatus = 'PASS';
    this.state.manifest.overall.status = 'PASS';

    // Sign manifest
    this.state.manifest.signature = this.signManifest();

    this.emit('releaseAuthorized', this.state.manifest);
    return true;
  }

  private generateManifest(): ReleaseManifest {
    const gates = Array.from(this.state.gates.values());
    
    const manifest: ReleaseManifest = {
      version: this.state.version,
      software: {},
      hardware: {},
      live: {},
      controls: {
        hardware_control: this.state.hardwareControl,
        raw_media_persistence: this.state.rawMediaPersistence
      },
      overall: {
        status: this.state.overallStatus,
        provenance_required: this.state.provenanceRequired
      }
    };

    // Populate software gates
    const softwareGates = gates.filter(g => 
      ['DEVELOPMENT', 'INTEGRATION', 'SECURITY', 'STAGING'].includes(g.stage)
    );
    softwareGates.forEach(g => {
      manifest.software[g.stage.toLowerCase()] = g.status;
    });

    // Populate hardware gates
    const hardwareGates = gates.filter(g => 
      ['H1', 'H2', 'H3', 'H4', 'H5', 'H6'].includes(g.stage)
    );
    hardwareGates.forEach(g => {
      manifest.hardware[`H${g.stage}`] = g.status;
    });

    // Populate live gates
    const liveGates = gates.filter(g => 
      ['LIVE', 'PARITY'].includes(g.stage)
    );
    liveGates.forEach(g => {
      manifest.live[g.stage.toLowerCase()] = g.status;
    });

    return manifest;
  }

  private signManifest(): string {
    const data = JSON.stringify({
      version: this.state.manifest.version,
      overall: this.state.manifest.overall,
      timestamp: new Date().toISOString()
    });
    return createHash('sha256').update(data).digest('hex');
  }

  // ============================================================
  // ROLLBACK
  // ============================================================

  public rollback(version: string): void {
    this.state.overallStatus = 'ROLLBACK_REQUIRED';
    this.state.hardwareControl = 'DISABLED';
    this.state.rawMediaPersistence = 'DISABLED';
    
    this.emit('rollbackInitiated', { 
      version, 
      timestamp: new Date().toISOString() 
    });
  }

  // ============================================================
  // STATE QUERIES
  // ============================================================

  public getState(): ReleaseState {
    return this.state;
  }

  public getManifest(): ReleaseManifest {
    return this.state.manifest;
  }

  public getOverallStatus(): GateStatus {
    return this.state.overallStatus;
  }

  public getEvidence(gateId: string): Evidence | undefined {
    const gate = this.state.gates.get(gateId);
    if (!gate || !gate.evidenceUri) return undefined;
    return this.evidenceStore.get(gate.evidenceUri);
  }

  public getIncidents(): Incident[] {
    return Array.from(this.incidentStore.values());
  }
}

// ============================================================
// SINGLETON EXPORT
// ============================================================

export const releaseGovernor = new ReleaseGovernor('3.0.0');
