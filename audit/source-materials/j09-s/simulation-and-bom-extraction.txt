    async def run_timeline(self, steps: int = 20):
        print(f"\n⏳ RUNNING {steps}-STEP TIMELINE SIMULATION...\n")
        for tick in range(steps):
            self.timeline = tick * 0.5
            self.orchestrator.tick = tick
            for bid, bearing in self.bearings.items():
                target_angle = (tick * np.pi / 18) % (2 * np.pi)
                await bearing.rotate_to(target_angle)
            if tick % 5 == 0 and tick > 0:
                origin = list(self.skater.nodes.keys())[0]
                dest = list(self.skater.nodes.keys())[-1]
                manifest = SkaterManifest(
                    id=f"SKATER_{tick}", cargo_type="RFF_FRAGMENT",
                    coherence_load=0.92, phase_signature=np.array([0.5, 0.5, 0.5])
                )
                result = self.skater.launch(manifest, origin, dest)
                print(f"   🚀 Skater {manifest.id}: {origin} → {dest} | {result['status']}")
                self.entities[manifest.id] = result
            self.emergence.tick()
            self.results.append({
                "tick": tick,
                "time_s": self.timeline,
                "network_stability": self.skater.network_health()["network_stability"],
                "emergence_coherence": self.emergence.global_coherence(),
                "bearing_health": [b.health_report()["health_score"] for b in self.bearings.values()]
            })
            await asyncio.sleep(0.01)
        print("\n✅ TIMELINE COMPLETE")

    def summary(self):
        final = self.results[-1] if self.results else {}
        print("\n📊 SIMULATION SUMMARY")
        print(f"   Final Network Stability: {final.get('network_stability',0):.3f}")
        print(f"   Final Coherence: {final.get('emergence_coherence',0):.3f}")
        print(f"   Entities Emerged: {len(self.entities)}")
        print(f"   Bearings Active: {len(self.bearings)}")
        return final

    async def shutdown(self):
        await self.orchestrator.shutdown()
