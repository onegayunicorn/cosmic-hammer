"""
J09 S — DIGITAL TWIN CORE
Sovereign Lattice Integration — v1.0 Gold Coast Baseline
Mirrors physical J09 S device state + extends into Lattice Entity
"""
import asyncio
import json
from dataclasses import dataclass, field, asdict
from typing import Dict, Optional, Callable, List
from enum import Enum
import time


class ButtonState(Enum):
    RELEASED = 0
    PRESSED = 1
    LONG_PRESS = 2
    DOUBLE_CLICK = 3


class TouchState(Enum):
    IDLE = 0
    TAP = 1
    SWIPE_LEFT = 2
    SWIPE_RIGHT = 3
    SWIPE_UP = 4
    SWIPE_DOWN = 5
    HOLD = 6


@dataclass
class J09SHardwareState:
    """Direct mirror of physical J09 S hardware"""
    connected: bool = False
    battery_percent: float = 100.0
    button_top: ButtonState = ButtonState.RELEASED
    button_mid: ButtonState = ButtonState.RELEASED
    button_bot: ButtonState = ButtonState.RELEASED
    touch_event: TouchState = TouchState.IDLE
    display_buffer: List[str] = field(default_factory=lambda: ["", "", ""])
    ble_rssi: int = -90
    last_packet_ms: float = 0.0
    blue_led: bool = False


@dataclass
class LatticeEntityState:
    """Sovereign Lattice Entity Extension"""
    entity_id: str = ""
    active: bool = False
    phase_rad: float = 0.0
    coherence: float = 0.92
    frequency: float = 432.0
    current_rff: Optional[str] = None
    rail_connected: bool = False
    timeline_azimuth: float = 0.0
    glyph_display: str = "⬡"


class J09SDigitalTwin:
    def __init__(self, config_path: str = "config.yaml"):
        self.load_config(config_path)
        self.hw = J09SHardwareState()
        self.entity = LatticeEntityState()
        self.entity.entity_id = f"J09_{int(time.time())}"
        self._callbacks: Dict[str, List[Callable]] = {}
        self._running = False
        self._sync_task = None

    def load_config(self, path: str):
        import yaml
        with open(path) as f:
            self.config = yaml.safe_load(f)

    # === STATE UPDATE — Physical → Twin ===
    def update_button(self, position: str, state: ButtonState):
        pos_map = {"top": "button_top", "mid": "button_mid", "bot": "button_bot"}
        setattr(self.hw, pos_map[position], state)
        self.hw.last_packet_ms = time.time()
        self._emit("button_event", position=position, state=state)
        # Auto-trigger Lattice actions
        if state == ButtonState.PRESSED:
            asyncio.create_task(self._handle_button_action(position))

    def update_touch(self, event: TouchState):
        self.hw.touch_event = event
        self.hw.last_packet_ms = time.time()
        self._emit("touch_event", event=event)

    def update_ble_status(self, connected: bool, rssi: int = None):
        self.hw.connected = connected
        if rssi: self.hw.ble_rssi = rssi
        self.hw.blue_led = connected
        self._emit("connection_event", connected=connected, rssi=rssi)
        if connected:
            self.entity.active = True
            self.entity.coherence = max(0.85, 1.0 - abs(rssi) / 100)

    def update_display(self, line1: str = "", line2: str = "", line3: str = ""):
        self.hw.display_buffer = [line1, line2, line3]
        self._emit("display_update", lines=self.hw.display_buffer)

    # === LATTICE ENTITY — Twin → Lattice ===
    def set_phase(self, phase_rad: float):
        self.entity.phase_rad = phase_rad % (2 * 3.14159)
        self.update_display(f"φ={self.entity.phase_rad*180/3.14:.0f}°",
                           f"Λ={self.entity.coherence:.2f}",
                           f"ƒ={self.entity.frequency:.0f}Hz")

    def load_rff(self, rff_id: str, glyph: str):
        self.entity.current_rff = rff_id
        self.entity.glyph_display = glyph
        self.update_display(glyph, f"RFF: {rff_id[:8]}", "LOADED")

    def manifest_rff(self) -> Dict:
        if not self.hw.connected or self.entity.coherence < 0.8:
            return {"status": "DENIED", "reason": "Not connected or low coherence"}
        self.update_display("⬡ MANIFEST", "TRIGGERED", "...")
        return {"status": "MANIFESTED", "rff": self.entity.current_rff,
                "phase": self.entity.phase_rad, "coherence": self.entity.coherence}

    # === CALLBACK SYSTEM ===
    def on(self, event: str, callback: Callable):
        if event not in self._callbacks:
            self._callbacks[event] = []
        self._callbacks[event].append(callback)

    async def _emit(self, event: str, **kwargs):
        if event in self._callbacks:
            for cb in self._callbacks[event]:
                if asyncio.iscoroutinefunction(cb):
                    await cb(**kwargs)
                else:
                    cb(**kwargs)

    async def _handle_button_action(self, position: str):
        """Map physical buttons to Lattice actions"""
        if position == "top":
            # Top Button → Load Next RFF
            await self._emit("load_next_rff")
        elif position == "mid":
            # Middle Button → Manifest Current RFF
            self.manifest_rff()
        elif position == "bot":
            # Bottom Button → Phase Lock / Compass Sync
            await self._emit("lock_azimuth", phase=self.entity.phase_rad)

    # === SYNC LOOP ===
    async def start_sync(self):
        self._running = True
        self._sync_task = asyncio.create_task(self._sync_loop())

    async def _sync_loop(self):
        while self._running:
            # Mirror state to Sovereign Lattice
            state = self.get_full_state()
            await self._emit("state_sync", state=state)
            await asyncio.sleep(self.config["twin"]["sync_interval_ms"] / 1000)

    def get_full_state(self) -> Dict:
        return {
            "hardware": asdict(self.hw),
            "entity": asdict(self.entity),
            "timestamp": time.time()
        }

    async def stop(self):
        self._running = False
        if self._sync_task:
            self._sync_task.cancel()
        self.hw.connected = False
        self.entity.active = False
        await self._emit("disconnected")

"""
INPUT AGENT — J09 S Button & Touch Processing
Learns patterns, debounces, triggers workflows
"""
import asyncio
from typing import Dict, List, Optional
from collections import deque
from core.digital_twin import J09SDigitalTwin, ButtonState, TouchState


class InputAgent:
    def __init__(self, twin: J09SDigitalTwin):
        self.twin = twin
        self.press_times: Dict[str, float] = {}
        self.touch_history = deque(maxlen=10)
        self.debounce_ms = 50
        self.long_press_ms = 500
        self.double_click_ms = 300
        self._setup_handlers()

    def _setup_handlers(self):
        self.twin.on("button_event", self._process_button)
        self.twin.on("touch_event", self._process_touch)

    async def _process_button(self, position: str, state: ButtonState):
        now = asyncio.get_event_loop().time()
        if state == ButtonState.PRESSED:
            self.press_times[position] = now
        elif state == ButtonState.RELEASED and position in self.press_times:
            duration = (now - self.press_times[position]) * 1000
            if duration < self.debounce_ms:
                return  # Debounce
            elif duration > self.long_press_ms:
                self.twin.update_button(position, ButtonState.LONG_PRESS)
                await self._on_long_press(position)
            elif position in self.press_times and (now - self.press_times.get(f"{position}_last", 0)) * 1000 < self.double_click_ms:
                self.twin.update_button(position, ButtonState.DOUBLE_CLICK)
                await self._on_double_click(position)
            setattr(self.press_times, f"{position}_last", now)
            del self.press_times[position]

    async def _process_touch(self, event: TouchState):
        self.touch_history.append((event, asyncio.get_event_loop().time()))
        # Gesture recognition
        if len(self.touch_history) >= 2:
            prev, _ = self.touch_history[-2]
            curr, _ = self.touch_history[-1]
            if prev == TouchState.TAP and curr == TouchState.TAP:
                await self._on_double_tap()

    async def _on_long_press(self, position: str):
        actions = {
            "top": ("SCAN_RFF", "Scanning Shard..."),
            "mid": ("MANIFEST", "Manifesting..."),
            "bot": ("CALIBRATE_COMPASS", "Calibrating...")
        }
        action, msg = actions.get(position, ("NONE", ""))
        self.twin.update_display(msg)
        await self.twin._emit("agent_action", action=action, source="long_press")

    async def _on_double_click(self, position: str):
        if position == "mid":
            await self.twin._emit("EMERGENCY_ABORT", source="double_click")

    async def _on_double_tap(self):
        await self.twin._emit("TOGGLE_ENTITY_MODE")

"""
BLE AGENT — J09 S Discovery, Pairing & HID Protocol
Auto-connect, reconnection, RSSI monitoring
"""
import asyncio
from typing import Optional, List
from bleak import BleakScanner, BleakClient


class BLEAgent:
    def __init__(self, twin, config):
        self.twin = twin
        self.config = config
        self.client: Optional[BleakClient] = None
        self.device_address = None
        self.retry_attempts = 0
        self.max_retries = 5
        self.services = {
            "HID": "00001812-0000-1000-8000-00805f9b34fb",
            "DEVICE_INFO": "0000180a-0000-1000-8000-00805f9b34fb",
            "BATTERY": "0000180f-0000-1000-8000-00805f9b34fb"
        }

    async def discover(self) -> List[Dict]:
        devices = await BleakScanner.discover(timeout=5)
        j09_devices = [d for d in devices if d.name and "J09" in d.name.upper()]
        return [{"name": d.name, "address": d.address, "rssi": d.rssi} for d in j09_devices]

    async def connect(self, address: str = None) -> Dict:
        address = address or self.config["device"]["ble_mac"]
        try:
            self.client = BleakClient(address, disconnected_callback=self._on_disconnect)
            await self.client.connect()
            self.device_address = address
            rssi = self.client.rssi if hasattr(self.client, 'rssi') else -60
            self.twin.update_ble_status(True, rssi)
            await self._setup_notifications()
            self.retry_attempts = 0
            return {"status": "CONNECTED", "address": address, "rssi": rssi}
        except Exception as e:
            self.retry_attempts += 1
            if self.retry_attempts < self.max_retries:
                await asyncio.sleep(1)
                return await self.connect(address)
            return {"status": "FAILED", "error": str(e)}

    async def _setup_notifications(self):
        for uuid in self.services.values():
            if self.client and self.client.is_connected:
                try:
                    await self.client.start_notify(uuid, self._on_data_received)
                except: pass

    def _on_data_received(self, sender, data):

            else: self.twin.update_button("top", ButtonState.RELEASED)
            if btn_byte & 0x02: self.twin.update_button("mid", ButtonState.PRESSED)
            else: self.twin.update_button("mid", ButtonState.RELEASED)

            else: self.twin.update_button("bot", ButtonState.RELEASED)

"""
LATTICE AGENT — J09 S ↔ Sovereign Lattice Integration
Connects physical device to Obsidian Shard, Chronal Compass, Rail Network
"""
import asyncio
from typing import Optional, Dict
from core.digital_twin import J09SDigitalTwin
from devices.obsidian_shard import ObsidianShard
from devices.chronal_compass import ChronalCompass


class LatticeAgent:
    def __init__(self, twin: J09SDigitalTwin):
        self.twin = twin
        self.shard: Optional[ObsidianShard] = None
        self.compass: Optional[ChronalCompass] = None
        self._setup_listeners()

    def _setup_listeners(self):
        self.twin.on("load_next_rff", self._load_next_rff)
        self.twin.on("lock_azimuth", self._lock_azimuth)
        self.twin.on("state_sync", self._sync_to_lattice)

    async def initialize_devices(self):
        self.shard = ObsidianShard()
        self.compass = ChronalCompass()
        await asyncio.gather(
            self.shard.power_on(),
            self.compass.power_on()
        )
        self.compass.pair_shard(self.shard.id)
        self.twin.entity.rail_connected = True
        self.twin.update_display("⬡ LATTICE", "ONLINE", "J09 READY")

    async def _load_next_rff(self):
        if not self.shard: return
        rffs = self.shard.list_rff()
        if not rffs:
            self.twin.update_display("NO RFF", "Stored", "")
            return
        # Cycle through RFFs
        current = self.twin.entity.current_rff
        idx = 0
        if current:
            ids = [r["id"] for r in rffs]
            idx = (ids.index(current) + 1) % len(ids) if current in ids else 0
        next_rff = rffs[idx]
        self.twin.load_rff(next_rff["id"], next_rff["glyph"])

    async def _lock_azimuth(self, phase: float):
        if not self.compass: return
        result = self.compass.lock_to_azimuth(phase)
        self.twin.update_display("🔒 LOCKED", f"φ={phase*180/3.14:.0f}°", "ANCHORED")

    async def _sync_to_lattice(self, state: Dict):
        """Sync Twin state → Lattice every 50ms"""
        if not self.shard or not self.twin.hw.connected:
            return
        # Update Shard metadata
        self.shard.metadata["j09_entity"] = state["entity"]
        # Update Compass coherence from BLE RSSI
        self.compass.coherence = self.twin.entity.coherence

"""

"""
import asyncio

async def rff_manifest_pipeline(twin: J09SDigitalTwin, rff_id: str = None):
    """
    FULL MANIFEST SEQUENCE:
    1. Mid Button Press → Trigger
    2. Load RFF from Shard
    3. Compass Lock to RFF Phase
    4. Update Display → Show Glyph + Status
    5. Sync to Lattice Orchestrator
    """
    # Stage 1 — Validate State
    if not twin.hw.connected:
        return {"status": "DENIED", "reason": "J09 S not connected"}
    if twin.entity.coherence < 0.8:
        return {"status": "DENIED", "reason": "Coherence below threshold (Λ ≥ 0.8)"}
    
    # Stage 2 — Load RFF
    if not rff_id:
        rff_id = twin.entity.current_rff
    if not rff_id:
        return {"status": "DENIED", "reason": "No RFF selected"}
    
    twin.update_display("⬡ LOADING", rff_id[:12], "...")
    await asyncio.sleep(0.3)
    
    # Stage 3 — Retrieve RFF Data
    from devices.obsidian_shard import ObsidianShard
    shard = ObsidianShard()
    shard.power_on()
    rff = shard.read_rff(rff_id)
    if not rff:
        return {"status": "NOT_FOUND", "rff_id": rff_id}
