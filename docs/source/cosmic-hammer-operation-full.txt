// Stochastic Success Model — Replace Certainty with Evidence-Based Probability

export interface SuccessFactors {
  technical: number; // T: 0-1 Technical feasibility
  market: number;     // M: 0-1 Market fit & adoption
  execution: number; // E: 0-1 Execution capability
  external: number; // X: 0-1 External/regulatory environment
  riskExposure: number; // R: 0-1 Downside risk exposure
}

export interface Coefficients {
  beta0: number; // Base log-odds (prior)
  beta1: number; // Technical weight
  beta2: number; // Market weight
  beta3: number; // Execution weight
  beta4: number; // External factor weight
  beta5: number; // Risk penalty weight
}

// Default coefficients — can be calibrated from historical data
export const DEFAULT_COEFFICIENTS: Coefficients = {
  beta0: -0.5, // Base prior: ~38% without factors
  beta1: 1.2,    // Technical feasibility is critical
  beta2: 1.0,    // Market fit is strong
  beta3: 0.8,    // Execution matters
  beta4: 0.5,    // External factors moderate
  beta5: 1.5     // Risk penalty is strongest
};

// Sigmoid — maps log-odds → probability (0, 1)
export const sigmoid = (z: number): number => 1 / (1 + Math.exp(-z));

// Calculate success probability — CORE FORMULA
export function calculateSuccessProbability(
  factors: SuccessFactors,
  coeffs: Coefficients = DEFAULT_COEFFICIENTS
): number {
  const { technical, market, execution, external, riskExposure } = factors;
  const { beta0, beta1, beta2, beta3, beta4, beta5 } = coeffs;

    const z = beta0 +
     beta1 * technical +
     beta2 * market +
     beta3 * execution +
     beta4 * external -
     beta5 * riskExposure;

    return sigmoid(z);
}

// Bernoulli trial — sample outcome from probability
export function sampleOutcome(p: number): boolean {
  return Math.random() < p;
}

// Confidence interval via Monte Carlo
export function probabilityBounds(
  factors: SuccessFactors,
  coeffs: Coefficients = DEFAULT_COEFFICIENTS,
  nSimulations = 10000,
  ciLevel = 0.9
): { mean: number; lower: number; upper: number } {
  const samples: number[] = [];
  for (let i = 0; i < nSimulations; i++) {
    // Add noise to factors to represent uncertainty
    const noisy = {
       technical: clamp(factors.technical + rnorm(0, 0.1)),
       market: clamp(factors.market + rnorm(0, 0.1)),
       execution: clamp(factors.execution + rnorm(0, 0.08)),
       external: clamp(factors.external + rnorm(0, 0.12)),
       riskExposure: clamp(factors.riskExposure + rnorm(0, 0.05))
     };
     samples.push(calculateSuccessProbability(noisy, coeffs));
    }
    samples.sort((a, b) => a - b);
    const lowerIdx = Math.floor(nSimulations * (1 - ciLevel) / 2);
    const upperIdx = Math.ceil(nSimulations * (1 + ciLevel) / 2);
    return {
      mean: samples.reduce((a, b) => a + b, 0) / nSimulations,
      lower: samples[lowerIdx],
      upper: samples[Math.min(upperIdx, nSimulations - 1)]
    };
}

// Helper: Normal random variable (Box-Muller)
function rnorm(mean = 0, sd = 1): number {
  const u1 = Math.random(), u2 = Math.random();
  const z = Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
  return mean + sd * z;
}

function clamp(x: number): number { return Math.max(0, Math.min(1, x)); }

 const { R_squared, deltaAIC, replications } = metrics;

    // Disproven: strongly worse than baseline

// Scientific Status Rules
export function determineScientificStatus(
  metrics: ValidationMetrics
): ScientificStatus {
  const { R_squared, deltaAIC, replications } = metrics;

 // Disproven: strongly worse than baseline
 if (deltaAIC > 10 && replications >= 2) {
   return 'DISPROVEN';
 }

 // Established: strong predictive power, beats baseline
 if (R_squared > 0.99 && deltaAIC < -10 && replications >= 3) {
   return 'ESTABLISHED';
 }

 // Experimental: moderate fit, some support
 if (R_squared > 0.8) {
   return 'EXPERIMENTAL';
 }

 // Speculative: low fit or unvalidated
 return 'SPECULATIVE';
}
